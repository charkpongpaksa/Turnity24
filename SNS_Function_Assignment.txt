Code:
import boto3
import os

sns = boto3.client("sns")

TOPIC_ARN = os.environ["TOPIC_ARN"]

def lambda_handler(event, context):
    alert_type = event.get("type", "assignment")

    if alert_type == "assignment":
        subject = "Assignment Alert"
        message = f"""
แจ้งเตือน Assignment

วิชา: {event.get("course", "-")}
งาน: {event.get("title", "-")}
Deadline: {event.get("deadline", "-")}

กรุณาตรวจสอบและส่งงานก่อนกำหนด
"""
    elif alert_type == "course":
        subject = "Course Alert"
        message = f"""
แจ้งเตือน Course

วิชา: {event.get("course", "-")}
หัวข้อ: {event.get("title", "-")}
รายละเอียด: {event.get("detail", "-")}

กรุณาเข้าไปตรวจสอบข้อมูลรายวิชา
"""
    else:
        subject = "Notification Alert"
        message = "มีการแจ้งเตือนใหม่"

    response = sns.publish(
        TopicArn=TOPIC_ARN,
        Subject=subject,
        Message=message
    )

    return {
        "statusCode": 200,
        "message": "sent",
        "type": alert_type,
        "messageId": response["MessageId"]
    }

import boto3
import os
from datetime import datetime, timezone

sns = boto3.client("sns")
TOPIC_ARN = os.environ["TOPIC_ARN"]

ASSIGNMENTS = [
    {
        "course": "Cloud Computing",
        "title": "AWS SNS Assignment",
        "deadline": "2026-04-30 23:59"
    },
    {
        "course": "Cloud Computing",
        "title": "Lambda + EventBridge Lab",
        "deadline": "2026-05-01 23:59"
    }
]

def lambda_handler(event, context):
    now = datetime.now()

    for assignment in ASSIGNMENTS:
        deadline = datetime.strptime(assignment["deadline"], "%Y-%m-%d %H:%M")
        hours_left = (deadline - now).total_seconds() / 3600

        if 0 < hours_left <= 24:
            message = f"""
แจ้งเตือนก่อน Deadline

วิชา: {assignment["course"]}
งาน: {assignment["title"]}
Deadline: {assignment["deadline"]}
เหลือเวลา: {int(hours_left)} ชั่วโมง

กรุณาส่งงานก่อนถึงกำหนด
"""

            sns.publish(
                TopicArn=TOPIC_ARN,
                Subject="Assignment Deadline Alert",
                Message=message
            )

    return {
        "statusCode": 200,
        "message": "Deadline check completed"
    }
